type t = Kernel.Ast.block =
  1. | Para of Inline.t node list
  2. | Section of t node list
    (*

    a heading (the first child) and the blocks up to the next heading of the same or a higher level. The heading's id is on the section node.

    *)
  3. | Heading of int * Inline.t node list
  4. | BlockQuote of t node list
  5. | CodeBlock of string * string
  6. | Div of string * t node list
    (*

    The string is a name, written ::: name where djot reads a class, an extension (Profile.with_ext_tags); empty means unnamed.

    *)
  7. | OrderedList of ordered_list_attributes * list_spacing * t node list node list
    (*

    Each item is a node, with its own location.

    *)
  8. | BulletList of char * list_spacing * t node list node list
    (*

    The character is the marker the items are written with: -, + or *, or : where definition lists are off.

    *)
  9. | TaskList of list_spacing * (task_status * t node list) node list
  10. | DefinitionList of list_spacing * (Inline.t node list node * t node list node) node list
    (*

    Each item holds its term and its definition.

    *)
  11. | ThematicBreak
  12. | Table of Inline.t node list node * cell node list node list
    (*

    Caption, then rows of cells. A caption with no inlines is no caption.

    *)
  13. | RawBlock of string * string
  14. | FootnoteDef of string * t node list
    (*

    A definition stays where it was written; Doc.footnotes is collected from these.

    *)
  15. | RefDef of string * string
  16. | Ext_keyed of Inline.t node list * t node
    (*

    An extension: label: content.

    *)
  17. | Ext_callout of string * callout_fold option * Inline.t node list * t node list
    (*

    Kind, fold marker, inline title, and body.

    *)
val to_string : ?profile:Profile.t -> ?style:style -> t node list -> string

The blocks as djot source, for the source of one node or of a mapped tree. Unlike Doc.to_string, a heading's identifier is always written as an attribute.

  • parameter profile

    the syntax to write; default Profile.djot
